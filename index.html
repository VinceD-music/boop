/* ============================================================ CLOUD CANVAS — animated storm background with parallax layers ============================================================ */ (function () { const canvas = document.getElementById('sky'); const ctx = canvas.getContext('2d'); const reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches; let W, H, DPR; function resize() { DPR = Math.min(window.devicePixelRatio || 1, 2); W = window.innerWidth; H = window.innerHeight; canvas.width = W * DPR; canvas.height = H * DPR; canvas.style.width = W + 'px'; canvas.style.height = H + 'px'; ctx.setTransform(DPR, 0, 0, DPR, 0, 0); } window.addEventListener('resize', resize); resize(); // Cloud "blobs" — soft radial gradients drifting at different speeds (parallax layers) function makeBlob(layer) { const speedBase = layer === 0 ? 0.06 : layer === 1 ? 0.12 : 0.2; const hueSets = [ ['rgba(14, 33, 82, 0.55)', 'rgba(14, 33, 82, 0)'], ['rgba(29, 77, 143, 0.5)', 'rgba(29, 77, 143, 0)'], ['rgba(79, 143, 209, 0.35)', 'rgba(79, 143, 209, 0)'], ['rgba(232, 160, 79, 0.12)', 'rgba(232, 160, 79, 0)'] ]; const colors = hueSets[Math.floor(Math.random() * hueSets.length)]; return { layer, x: Math.random() * W, y: Math.random() * H, r: (200 + Math.random() * 320) * (1 + layer * 0.4), vx: (Math.random() - 0.5) * speedBase, vy: (Math.random() - 0.5) * speedBase * 0.4, colors }; } let blobs = []; function initBlobs() { blobs = []; const counts = [4, 5, 4]; // back, mid, front layers counts.forEach((count, layer) => { for (let i = 0; i < count; i++) blobs.push(makeBlob(layer)); }); } initBlobs(); // Scroll-driven drift offset — adds extra motion when changing sections let scrollDrift = 0; let targetDrift = 0; function draw() { ctx.clearRect(0, 0, W, H); // base sky gradient const base = ctx.createRadialGradient(W * 0.5, H * 0.15, 0, W * 0.5, H * 0.5, Math.max(W, H) * 0.9); base.addColorStop(0, '#10295c'); base.addColorStop(0.45, '#081636'); base.addColorStop(1, '#050b1a'); ctx.fillStyle = base; ctx.fillRect(0, 0, W, H); blobs.forEach((b) => { if (!reduceMotion) { b.x += b.vx + scrollDrift * (0.15 + b.layer * 0.1); b.y += b.vy; if (b.x < -b.r) b.x = W + b.r; if (b.x > W + b.r) b.x = -b.r; if (b.y < -b.r) b.y = H + b.r; if (b.y > H + b.r) b.y = -b.r; } const g = ctx.createRadialGradient(b.x, b.y, 0, b.x, b.y, b.r); g.addColorStop(0, b.colors[0]); g.addColorStop(1, b.colors[1]); ctx.fillStyle = g; ctx.beginPath(); ctx.arc(b.x, b.y, b.r, 0, Math.PI * 2); ctx.fill(); }); // ease drift back to baseline scrollDrift += (targetDrift - scrollDrift) * 0.08; requestAnimationFrame(draw); } draw(); // Trigger extra drift burst on section change (wheel/scroll) let lastScrollY = window.scrollY; let driftTimeout; window.addEventListener('scroll', () => { const delta = window.scrollY - lastScrollY; lastScrollY = window.scrollY; targetDrift = Math.max(-2.5, Math.min(2.5, delta * 0.08)); clearTimeout(driftTimeout); driftTimeout = setTimeout(() => { targetDrift = 0; }, 400); }, { passive: true }); })(); /* ============================================================ NAV — background on scroll ============================================================ */ const nav = document.getElementById('nav');
